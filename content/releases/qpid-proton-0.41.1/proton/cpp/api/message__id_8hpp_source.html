96template <> struct std::hash<proton::message_id> {
97 std::size_t operator()(const proton::message_id& s) const {
98 std::size_t h;
99 auto type = s.type();
100 switch(type) {
101 case proton::ULONG: h = std::hash<uint64_t>{}(proton::internal::get<uint64_t>(s)); break;
102 case proton::UUID: h = std::hash<proton::uuid>{}(proton::internal::get<proton::uuid>(s)); break;
103 case proton::BINARY: h = std::hash<proton::binary>{}(proton::internal::get<proton::binary>(s)); break;
104 case proton::STRING: h = std::hash<std::string>{}(proton::internal::get<std::string>(s)); break;
105 default: throw proton::conversion_error("invalid message_id type "+type_name(s.type()));
106 }
107 auto type_hash = std::hash<proton::type_id>{}(type);
108 // Combine the two hashes - see https://stackoverflow.com/a/5889238
109 return h ^ (type_hash + 0x9e3779b9 + (h << 6) + (h >> 2));
110}
